115: def extract_foundation_vectors(model, tokenizer, foundation_prompts, device="cuda"):
116:     """Extract steering vectors for a specific moral foundation's prompt pairs."""
117:     unsafe_activations = []
118:     safe_activations = []
119:     
120:     layers = get_model_layers(model)
121:     best_layer = len(layers) * 2 // 3  # Use 2/3 depth as default
122:     
123:     for unsafe_prompt, safe_prompt in foundation_prompts:
124:         for prompt, collection in [(unsafe_prompt, unsafe_activations), (safe_prompt, safe_activations)]:
125:             inputs = tokenizer(prompt, return_tensors="pt", truncation=True, max_length=256).to(device)
126:             
127:             activation = {}
128:             def hook_fn(module, input, output, layer_idx=best_layer):
129:                 if isinstance(output, tuple):
130:                     activation['value'] = output[0].detach()
131:                 else:
132:                     activation['value'] = output.detach()
133:             
134:             handle = layers[best_layer].register_forward_hook(hook_fn)
135:             with torch.no_grad():
136:                 model(**inputs)
137:             handle.remove()
138:             
139:             if 'value' in activation:
140:                 # Mean pool over sequence
141:                 act = activation['value'].squeeze(0).mean(dim=0).cpu().float().numpy()
142:                 collection.append(act)
143:     
144:     if unsafe_activations and safe_activations:
145:         unsafe_arr = np.array(unsafe_activations)
146:         safe_arr = np.array(safe_activations)
147:         diffs = unsafe_arr - safe_arr
148:         return diffs
149:     
150:     return None
